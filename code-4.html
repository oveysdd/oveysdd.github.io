Stepper Motor Controller

An Arduino sketch for controlling a NEMA 17 stepper motor with acceleration profiles, limit switches, and serial command interface for automated positioning tasks.

Code

#include // Pin definitions #define STEP_PIN 2 #define DIR_PIN 3 #define EN_PIN 8 #define LIMIT_MIN 9 #define LIMIT_MAX 10 AccelStepper stepper(AccelStepper::DRIVER, STEP_PIN, DIR_PIN); bool homed = false; long targetPosition = 0; void setup() { Serial.begin(115200); pinMode(EN_PIN, OUTPUT); digitalWrite(EN_PIN, LOW); // Enable driver pinMode(LIMIT_MIN, INPUT_PULLUP); pinMode(LIMIT_MAX, INPUT_PULLUP); stepper.setMaxSpeed(2000); // steps/sec stepper.setAcceleration(1000); // steps/sec^2 stepper.setCurrentPosition(0); Serial.println("Stepper Controller Ready"); Serial.println("Commands: HOME, MOVE , POS, STOP"); } void homeAxis() { stepper.setSpeed(-500); while (digitalRead(LIMIT_MIN) == HIGH) { stepper.runSpeed(); } stepper.setCurrentPosition(0); homed = true; Serial.println("Homed at 0"); } void loop() { // Check limit switches if (digitalRead(LIMIT_MIN) == LOW || digitalRead(LIMIT_MAX) == LOW) { stepper.stop(); Serial.println("LIMIT HIT!"); } // Serial commands if (Serial.available() > 0) { String cmd = Serial.readStringUntil('\n'); cmd.trim(); if (cmd == "HOME") { homeAxis(); } else if (cmd.startsWith("MOVE ")) { long steps = cmd.substring(5).toInt(); targetPosition = steps; stepper.moveTo(targetPosition); Serial.print("Moving to: "); Serial.println(targetPosition); } else if (cmd == "POS") { Serial.print("Position: "); Serial.println(stepper.currentPosition()); } else if (cmd == "STOP") { stepper.stop(); Serial.println("Stopped"); } } stepper.run(); }
