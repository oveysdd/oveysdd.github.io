Modal Analysis Solver

A lightweight C implementation for extracting natural frequencies and mode shapes from mass and stiffness matrices using the power iteration method.

Code

#include #include #define N 3 #define EPS 1e-6 #define MAX_ITER 1000 void mat_vec_mult(double A[N][N], double v[N], double result[N]) { for (int i = 0; i < N; i++) { result[i] = 0; for (int j = 0; j < N; j++) { result[i] += A[i][j] * v[j]; } } } double vec_norm(double v[N]) { double sum = 0; for (int i = 0; i < N; i++) sum += v[i] * v[i]; return sqrt(sum); } double power_iteration(double A[N][N], double eigenvec[N]) { double v[N] = {1, 1, 1}; double v_new[N]; double lambda = 0; for (int iter = 0; iter < MAX_ITER; iter++) { mat_vec_mult(A, v, v_new); double norm = vec_norm(v_new); for (int i = 0; i < N; i++) v_new[i] /= norm; double lambda_new = 0; mat_vec_mult(A, v_new, v); for (int i = 0; i < N; i++) lambda_new += v_new[i] * v[i]; if (fabs(lambda_new - lambda) < EPS) break; lambda = lambda_new; for (int i = 0; i < N; i++) v[i] = v_new[i]; } for (int i = 0; i < N; i++) eigenvec[i] = v[i]; return lambda; } int main() { double K[N][N] = {{2, -1, 0}, {-1, 2, -1}, {0, -1, 1}}; double M[N][N] = {{1, 0, 0}, {0, 1, 0}, {0, 0, 1}}; double mode[N]; // Simplified: solve K*phi = lambda*M*phi double lambda = power_iteration(K, mode); double freq = sqrt(lambda) / (2 * M_PI); printf("Natural Frequency: %.4f Hz\n", freq); printf("Mode Shape: [%.3f, %.3f, %.3f]\n", mode[0], mode[1], mode[2]); return 0; }
